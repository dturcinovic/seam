package out // import "github.com/dturcinovic/seam/internal/app/port/out"

Package out holds the outgoing ports, cut by capability, which the use cases
call and the driven adapters implement. It knows nothing but the domain.

TYPES

type BlobReader interface {
	ReadBlob(ctx context.Context, id change.ObjectID, max limits.FileSize) ([]byte, error)
}
    BlobReader reads a blob of the repository. It never reads more than max+1
    bytes: a larger blob yields *change.TooLargeError, an object that is no blob
    change.ErrNotRegular, a missing one change.ErrNotExist.

type ChangeLister interface {
	ListChanges(ctx context.Context, c change.Comparison) ([]change.ChangedFile, error)
}
    ChangeLister lists the files that differ between the sides of a resolved
    comparison, as git sees them; files git does not track are not among them.

type ChangeWatcher interface {
	// Watch watches git's index and HEAD in p, and with worktree the files
	// under p's root but its ignored directories, until ctx ends, and tells
	// l. It returns nil once ctx ends and an error if watching fails, at
	// the start or later.
	Watch(ctx context.Context, p change.WatchPaths, worktree bool, l WatchListener) error
}
    ChangeWatcher watches the files of a repository for changes.

type CheckoutRoots interface {
	// Mount makes dir the tree of commit id's documents.
	Mount(id change.ObjectID, dir string)
}
    CheckoutRoots tells the language servers where the checkout of a commit
    lies, so documents of that commit can be served from it.

type CommitCheckouts interface {
	// Checkout returns the directory of a checkout of commit id, making it
	// the first time. That takes as long as git takes to write the tree.
	Checkout(ctx context.Context, id change.ObjectID) (dir string, err error)
}
    CommitCheckouts makes checkouts of commits for the session, which go with
    it.

type DefinitionRequester interface {
	RequestDefinition(p hover.Position) hover.Submit
	// CancelDefinition drops the request: no answer comes for it.
	CancelDefinition(id hover.ReqID)
}
    DefinitionRequester asks where a symbol is defined. Neither call blocks;
    the answer comes through ServerEvents as DefinitionDone.

type DocumentSync interface {
	// Sync opens d or brings it to d's version, starting its server if need
	// be. It names the instance that serves d; ok is false if no server
	// hovers over d's language, or d's commit has no checkout mounted.
	Sync(d hover.Document) (i lang.Instance, ok bool)
	// Close closes the document k; its server forgets the text.
	Close(k hover.DocKey)
}
    DocumentSync tells the language servers which documents are open and with
    what text. Neither call blocks or is refused; the text reaches the server
    once it can take it, the newest version only.

type Highlighter interface {
	Highlight(ctx context.Context, l lang.Lang, text string) (t syntax.Tokens, ok bool)
}
    Highlighter finds the tokens of text in language l, text that holds no
    line over syntax.MaxLine. It covers the lines from the first up to where
    it stops: when ctx ends, which it checks between tokens, or when the lexer
    fails. ok is false if it has no lexer for l.

type HoverRequester interface {
	Request(p hover.Position) hover.Submit
	// Cancel drops the request: no answer comes for it.
	Cancel(id hover.ReqID)
}
    HoverRequester asks for hovers. Neither call blocks; the answer comes
    through ServerEvents.

type IgnoreChecker interface {
	// Ignored returns those of paths, absolute and in the worktree, that git
	// ignores.
	Ignored(ctx context.Context, paths []string) ([]string, error)
}
    IgnoreChecker asks git what it ignores.

type PathLister interface {
	ListPaths(ctx context.Context, left, right string) ([]change.ChangedFile, error)
}
    PathLister lists the files that differ between two directories on disk,
    with paths relative to each. A symlink counts as the file it points to:
    git difftool -d puts the worktree side there as links into the worktree.

type PathReader interface {
	ReadPath(ctx context.Context, path string, max limits.FileSize) ([]byte, change.Stamp, error)
	// StatPath returns the stamp of the file without reading it.
	StatPath(ctx context.Context, path string) (change.Stamp, error)
}
    PathReader reads a file on disk, following symlinks. Like WorktreeReader
    it returns the stamp of what it read, never reads more than max+1 bytes and
    never blocks on a special file.

type ReleaseFetcher interface {
	LatestRelease(ctx context.Context) (release.Version, error)
}
    ReleaseFetcher asks where seam is published for its latest release.

type ReleaseStore interface {
	LoadCheck(ctx context.Context) (release.Check, error)
	SaveCheck(ctx context.Context, c release.Check) error
}
    ReleaseStore keeps the last check for a newer release between sessions.
    A check never saved loads as the zero Check, one that cannot be read as the
    zero Check with the error.

type RevisionResolver interface {
	// ResolveCommit fails for a revision that names no commit.
	ResolveCommit(ctx context.Context, rev string) (change.ObjectID, error)
}
    RevisionResolver turns a revision as the user names it into a commit.

type ServerEvents interface {
	Signal() <-chan struct{}
	Drain() []hover.Update
}
    ServerEvents hands out the answers and status changes of the servers.
    Signal is set after each new one; Drain returns them, oldest first.

type SubmoduleReader interface {
	ReadSubmodule(ctx context.Context, path string) (commit change.ObjectID, dirty bool, err error)
}
    SubmoduleReader reads a submodule of the worktree as git diff sees it:
    the commit it is at, and dirty if its worktree has changes; untracked files
    do not count. A submodule that is not checked out yields change.ErrNotExist.

type UntrackedLister interface {
	ListUntracked(ctx context.Context) ([]string, error)
}
    UntrackedLister lists the worktree files git does not track and does not
    ignore, relative to the repository root. An embedded repository (a nested
    `git init`, a linked worktree) is no file and is not listed.

type WatchListener interface {
	// Changed reports a change, without saying which, and once when the
	// watcher is set up, as what changed before cannot be told. It must not
	// block.
	Changed()
	// Ignored returns those of dirs, directories that came into the
	// worktree after the start, that git ignores; the watcher leaves them
	// out. The watcher waits for the answer.
	Ignored(dirs []string) []string
}
    WatchListener hears from a ChangeWatcher. Its calls come from the watcher's
    goroutine, one at a time.

type WatchPathsReader interface {
	// WatchPaths returns the root, git's directory and the directories git
	// ignores, as they are now.
	WatchPaths(ctx context.Context) (change.WatchPaths, error)
}
    WatchPathsReader names what a watcher of the repository looks at.

type WorktreeLocator interface {
	// LocateInWorktree returns the path, its symlinks followed, relative to
	// the worktree's root; ok is false outside it, in .git, or without one.
	LocateInWorktree(path string) (rel string, ok bool)
}
    WorktreeLocator finds a file on disk in the repository's worktree, where a
    language server can work on it.

type WorktreeReader interface {
	ReadWorktree(ctx context.Context, path string, max limits.FileSize) ([]byte, change.Stamp, error)
}
    WorktreeReader reads a file of the worktree with the stamp of what it read.
    A symlink yields its target as text, as git stores it. It never reads
    more than max+1 bytes and never blocks on a special file: a larger file
    yields *change.TooLargeError, anything but a regular file or symlink
    change.ErrNotRegular, a missing path change.ErrNotExist.

type WorktreeStatter interface {
	StatWorktree(ctx context.Context, path string) (change.Stamp, error)
}
    WorktreeStatter returns the stamp of a worktree file without reading it.
