package note // import "github.com/dturcinovic/seam/internal/domain/note"

Package note holds the review notes set at places of a diff: what a note is,
where it is anchored, how its place is found again in a later version of the
file, and the book that keeps them all. It reads and writes no file and knows no
format of storage.

CONSTANTS

const (
	// MaxReply is the most bytes of an answer.
	MaxReply = 64 << 10
	// MaxTake is the most notes taken at once; more are left for the next
	// call.
	MaxTake = 20
)
    Limits of what agents take and answer.

const (
	// MaxText is the most bytes of a note's text.
	MaxText = 8 << 10
	// MaxLines is the most lines a selection holds.
	MaxLines = 200
	// MaxNotes is the most notes not resolved a book holds. Resolved ones
	// do not count, as nothing removes them; MaxFile bounds them all.
	MaxNotes = 2000
	// MaxFile is the most bytes the stored book may take.
	MaxFile = 16 << 20
)
    Limits of the book; anything beyond is refused with a named error.

const MaxBatches = 64
    MaxBatches is the most batches a book keeps; the oldest go first, and taking
    by one gone fails with ErrNoBatch.


VARIABLES

var (
	ErrReplyTooLong = errors.New("answer too long")
	ErrNotTaken     = errors.New("note not taken")
	ErrNoSession    = errors.New("no such agent session")
)
    Errors of agent sessions taking and answering notes.

var (
	ErrNothingToSend = errors.New("no notes to send")
	ErrNoBatch       = errors.New("no such batch")
)
    Errors of sending notes to agent sessions.

var (
	ErrTextTooLong  = errors.New("note text too long")
	ErrEmptyText    = errors.New("note text empty")
	ErrTooManyLines = errors.New("selection too long")
	ErrTooManyNotes = errors.New("too many notes")
	ErrBookTooLarge = errors.New("notes file too large")
	ErrNoNote       = errors.New("no such note")
	ErrDuplicateID  = errors.New("note id taken")
	ErrInvalidRange = errors.New("invalid selection")
	ErrInvalidKind  = errors.New("invalid note kind")
	ErrNoOrigin     = errors.New("note anchored in no commit, index or worktree")
	// ErrResolved is an answer to a note already resolved.
	ErrResolved = errors.New("note resolved")
	// ErrNotesDamaged is a stored book that cannot be read; it is never
	// written over, as that would lose the notes in it.
	ErrNotesDamaged = errors.New("notes file damaged")
	// ErrNotesVersion is a stored book of a version this seam does not
	// know; it is never written over either.
	ErrNotesVersion = errors.New("notes file of unknown version")
	// ErrNoBook is notes asked for outside a repository.
	ErrNoBook = errors.New("no notes outside a repository")
	// ErrNotesFailed answers every change once the notes' background
	// work has panicked and runs no more.
	ErrNotesFailed = errors.New("notes failed")
)
    Errors of the book and of its storage.


TYPES

type Anchor struct {
	Path            string
	Origin          Origin
	Blob            change.ObjectID
	Start, End      int
	Lines           []string
	Before, After   []string
	Copies, Ordinal int
}
    Anchor is where a note is set: the lines Start to End of Path, 1-based and
    End included, in the version Blob of Origin. Lines is the selection as it
    stood, kept unchanged for the agent; Before and After are up to three lines
    around it, to find it again. Copies is how often selection and context stood
    in that version, and Ordinal which of them is meant.

func NewAnchor(path string, origin Origin, blob change.ObjectID, lines []string, start, end int) (Anchor, error)
    NewAnchor anchors the lines start to end of lines, the version blob of path
    in origin.

type Answer struct {
	Reply ID
	Late  bool
}
    Answer is what came of a reply: the reply's ID, an earlier one's for the
    same reply again, and whether it came late, under a claim no longer the
    note's.

type Batch struct {
	ID      ID
	Session ID
	Notes   []ID
	At      time.Time
}
    Batch is a set of notes sent to one agent session.

type Book struct {
	Notes    []Note
	Sessions []Session
	Batches  []Batch
}
    Book holds all notes of a worktree, with the sessions and batches of the
    agents that read them.

func (b *Book) Add(d Draft, id ID, at time.Time) error
    Add takes d as a new open note named id, set at at.

func (b *Book) Join(s Session) error
    Join adds the agent session s.

func (b *Book) Leave(id ID)
    Leave removes session id and gives back the notes it took, unanswered.
    A session gone already is no error, so leaving twice does no harm.

func (b *Book) Prune(alive func(pid int) bool)
    Prune removes the sessions whose process is no longer alive. A session that
    holds a claim stays: its notes show who took them until the user gives them
    back.

func (b *Book) Release(id ID) error
    Release gives note id back from the session that took it, unanswered.

func (b *Book) Reply(id ID, r Reply) (Answer, error)
    Reply adds r to note id. The same text under the same claim again adds
    nothing and answers with the reply there, so that an agent may repeat a
    reply whose answer it lost. A reply under a claim other than the note's
    comes late: it is kept, but the newer taking stands. Any other answers the
    note.

func (b *Book) Resolve(id ID) error
    Resolve marks note id done; a note resolved already stays so.

func (b *Book) Send(id, session ID, at time.Time, alive func(pid int) bool) (Batch, error)
    Send sends the waiting notes, open or sent and not taken, to session in a
    new batch named id, set at at. The batch holds these notes alone: a note set
    later waits for the next one. A note sent before moves to the new batch,
    so a notice that got lost is made up for without a note being taken twice.
    Session must be one of the book's Targets: a session whose process died
    would never take the batch.

func (b *Book) Sendings(session ID) []Sending
    Sendings are the batches sent to session that still have notes sent and not
    taken, oldest first.

func (b *Book) Session(id ID) (Session, bool)
    Session is the session named id.

func (b *Book) Take(ids []ID, c Claim) (Taking, error)
    Take takes notes under c, for its session: those of ids, or without ids
    the waiting ones, open or sent, at most MaxTake either way, and More says
    waiting ones are left. Only waiting notes are taken, whatever session their
    batch was sent to; the others of ids come back refused.

func (b *Book) TakeBatch(id ID, c Claim) (Taking, error)
    TakeBatch takes under c the notes of batch id still sent in it, at most
    MaxTake, if it was sent to c's session. The others come back refused:
    all of a batch sent to another session, and those sent again since in a
    newer batch.

func (b *Book) Targets(alive func(pid int) bool) []Session
    Targets are the sessions whose process runs, the ones notes are sent to.
    A dead session that holds a claim stays in the book but is none.

type Claim struct {
	ID      ID
	Session ID
	At      time.Time
}
    Claim is an agent session having taken a note.

type Draft struct {
	Kind       Kind
	Anchor     Anchor
	Comparison string
	Text       string
}
    Draft is a note as the user writes it, before the book takes it.

type Head struct {
	ID     ID
	Path   string
	Origin Origin
	Kind   Kind
	State  State
}
    Head is a note in brief, as the file list and the footer count them.

type ID string
    ID names a note, a reply, a claim, a batch or a session.

type Kind uint8
    Kind is what a note asks for.

const (
	Question Kind
	Finding
)
    The kinds of note. The zero value is no valid kind.

func (k Kind) String() string
    String is the kind's name in storage.

type Note struct {
	ID         ID
	Kind       Kind
	Anchor     Anchor
	Comparison string
	Text       string
	State      State
	Created    time.Time
	// Claim is the taking of a note Taken; Batch belongs to sending it,
	// which the book keeps as it is.
	Claim   *Claim
	Batch   ID
	Replies []Reply
}
    Note is a question or finding at an anchor. Comparison describes the
    comparison it was set in, for people and agents only: the note belongs to
    the anchor's origin.

type Op uint64
    Op names a change to the book asked for; its outcome comes as Done.

type Origin struct {
	Kind   change.SourceKind
	Commit change.ObjectID
}
    Origin is where the text a note is set on comes from: a commit, the index or
    the worktree. A note shows in every comparison with its origin on a side.

func OriginOf(s change.Source) (Origin, bool)
    OriginOf is the origin of side s; ok is false for a file on disk, which is
    no place for notes.

func (o Origin) String() string
    String names the origin as the user would.

type Placement struct {
	Where      Where
	Start, End int
}
    Placement is where a note stands in a version of its file: Start to End,
    1-based and End included, when Where is Shown.

func Locate(a Anchor, blob change.ObjectID, lines []string) Placement
    Locate finds a's place in lines, the version blob of a's file. In the
    version it was set in it stands where it was set. Elsewhere it is a best
    guess from the text around it, no proof of identity: a block pasted in with
    the same selection and context cannot be told from the original. Rather than
    guess between places, it answers Ambiguous.

type Reason string
    Reason is why a note asked for was not taken; agents read it as is.

const (
	RefusedUnknown  Reason = "unknown"
	RefusedTaken    Reason = "taken"
	RefusedAnswered Reason = "answered"
	RefusedResolved Reason = "resolved"
)
    The reasons a note is not taken. Only waiting notes are, open or sent:
    a question that remains after an answer is a new note.

const (
	// RefusedOtherSession is a batch sent to another session.
	RefusedOtherSession Reason = "other_session"
	// RefusedRebatched is a note sent again since, in a newer batch.
	RefusedRebatched Reason = "rebatched"
)
    The reasons a note of a batch is not taken, beside those of Take.

type Refusal struct {
	ID     ID
	Reason Reason
}
    Refusal is a note asked for and not taken.

type Reply struct {
	ID     ID
	Author string
	Claim  ID
	Text   string
	At     time.Time
}
    Reply is an answer to a note.

type Sending struct {
	Batch Batch
	Notes []Note
}
    Sending is a batch with those of its notes still sent, not taken.

type Session struct {
	ID      ID
	Client  string
	PID     int
	Started time.Time
}
    Session is an agent session that reads notes.

func TakenBy(n Note, sessions []Session) (Session, bool)
    TakenBy is the session among sessions that holds n's claim; ok is false for
    a note no session holds, or one whose session is gone.

type SnapshotKey struct {
	Path   string
	Origin Origin
	Blob   change.ObjectID
}
    SnapshotKey names a side of a file as shown: its path, origin and blob.
    The blob names the text, so a placement that comes late is still right for
    every side with the same key.

type Spot struct {
	Note  Note
	Place Placement
}
    Spot is a note and its place in a side.

type State uint8
    State is where a note stands.

const (
	Open State
	Resolved
	// Taken is a note an agent session took and has not answered yet.
	Taken
	// Answered is a note an agent answered.
	Answered
	// Sent is a note sent to an agent session in a batch, which no session
	// took yet.
	Sent
)
    The states. The zero value is no valid state.

func (s State) String() string
    String is the state's name in storage.

func (s State) Waits() bool
    Waits reports that a note in s waits for an agent to take it: open, or sent
    and not taken.

type Taking struct {
	Claim   Claim
	Notes   []Note
	Refused []Refusal
	More    bool
}
    Taking is what one Take took: the notes under Claim, the notes asked for and
    refused, and whether open notes are left past MaxTake.

type Update struct {
	Kind UpdateKind

	Key   SnapshotKey
	Spots []Spot

	Op  Op
	Err error

	Heads    []Head
	Sessions []Session
	Targets  []Session
	Live     bool

	Notes []Note
}
    Update is news from the notes.

type UpdateKind uint8
    UpdateKind tells which fields of an Update are set.

const (

	// Placed sets Key and Spots: the notes of that side and where they
	// stand.
	Placed UpdateKind
	// Done sets Op and, if it failed, Err.
	Done
	// StoreStatus sets Heads, all notes in brief, Sessions, the agent
	// sessions that took notes or may, Targets, those of them whose
	// process runs, to send notes to, and Err, why the book cannot be read
	// or is not watched, nil once it is again. Live is whether changes
	// from elsewhere show.
	StoreStatus
	// Overview sets Notes, every note of the book, resolved ones too, or
	// Err, why there is none.
	Overview
)
    The kinds of update. The zero value is no valid kind.

type Where uint8
    Where is how surely a note was found again.

const (

	// Exact is the selection with its full context at exactly one place,
	// of a selection that stood once.
	Exact Where
	// Likely is the best guess, but no proof: the one place with at least
	// half the context, or the copy by its order among as many as there
	// were.
	Likely
	// Changed is the lines between the context before and after, each
	// found once: the selection was changed within.
	Changed
	// Ambiguous is more than one place, none better than the others.
	Ambiguous
	// Orphaned is no place.
	Orphaned
)
    The outcomes of Locate. The zero value is no outcome.

func (w Where) Shown() bool
    Shown reports that the outcome has a place to show the note at.

func (w Where) String() string
    String names the outcome for the user.
