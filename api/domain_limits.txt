package limits // import "github.com/dturcinovic/seam/internal/domain/limits"

Package limits holds the bounds seam keeps on what it reads and keeps in memory,
as types that check themselves. It does not enforce them; the readers and caches
that take them do.

VARIABLES

var ErrInvalidFileSize = errors.New("file size limit must be a positive size")
    ErrInvalidFileSize is returned for a limit that is not a positive size.

var ErrInvalidInstances = errors.New("instance limit out of bounds")
    ErrInvalidInstances is returned for a limit outside 1 to MaxInstances.


TYPES

type FileSize int64
    FileSize is the most bytes one side of a file may have to be loaded.
    The zero value is not a valid limit; use NewFileSize or DefaultFileSize.

const DefaultFileSize FileSize = 8 << 20
    DefaultFileSize is the limit when none is configured. Up to this size a file
    gets the full treatment: diff, inline diff and highlighting.

func NewFileSize(n int64) (FileSize, error)
    NewFileSize checks n and returns it as a limit.

func ParseFileSize(s string) (FileSize, error)
    ParseFileSize reads a limit as written by a user: bytes, or a number with a
    binary suffix K, M or G, such as 16M.

func (s FileSize) Bytes() int64
    Bytes is the limit as a byte count.

type Instances int
    Instances is how many language server instances may run at once. The zero
    value is not a valid limit; use NewInstances or DefaultInstances.

const (
	DefaultInstances Instances = 4
	MaxInstances     Instances = 16
)
    The bounds of Instances. Each instance is a server process with its own
    index of a workspace, which can take gigabytes.

func NewInstances(n int) (Instances, error)
    NewInstances checks n and returns it as a limit.

func (n Instances) OrDefault() Instances
    OrDefault is n, or DefaultInstances if n is not set.
