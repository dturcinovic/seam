package in // import "github.com/dturcinovic/seam/internal/app/port/in"

Package in holds the incoming ports, which the TUI adapter calls and the use
cases implement. It knows nothing but the domain.

TYPES

type ChangesetLoader interface {
	// LoadChangeset resolves the comparison's commits and returns them with
	// the changed files, untracked ones included when the right side is the
	// worktree.
	LoadChangeset(ctx context.Context, c change.Comparison) (change.Changeset, error)
}
    ChangesetLoader loads the files that differ in a comparison. It is called
    once, before the TUI starts; an error ends seam.

type ChangesetReloader interface {
	// Reload starts a reload and replaces one still waiting.
	Reload(gen change.Gen)
	// Signal is set, without blocking, after each new update.
	Signal() <-chan struct{}
	// Drain hands over the updates since the last call, oldest first.
	Drain() []change.Update
}
    ChangesetReloader reloads the changeset while the TUI runs, resolving the
    revisions of the first load anew, as a commit or a checkout moves HEAD.
    Reload never blocks: the work runs in the background, and its outcome waits
    in Drain until the UI collects it.

type Definer interface {
	// Define asks at p, as Hover does; only the answer to the last p asked
	// is handed on.
	Define(p hover.Position) hover.Submit
	// CancelDefinition drops the request: no answer comes for it.
	CancelDefinition(id hover.ReqID)
}
    Definer asks the language server where the symbol at a position is defined.
    Neither call blocks; the answer comes through LSPUpdates.

type DocumentTracker interface {
	// Track opens d or brings it to d's version, and keeps it open while it
	// is among the last shown. It names the instance that serves d; ok is
	// false if none does, as for a document without a path. A document of
	// a commit not checked out yet is served by the checkout's instance
	// until the checkout is Ready; Track then names the server's.
	Track(d hover.Document) (i lang.Instance, ok bool)
	// Untrack closes the document k: the file is gone, or its text no
	// longer what the viewer shows, and the server reads it from disk.
	Untrack(k hover.DocKey)
}
    DocumentTracker tells the language servers which documents the viewer shows.
    Neither call blocks or is refused.

type FileDiffLoader interface {
	// Load asks for a file. It replaces a request still waiting and cancels
	// the running one if that is for another file.
	Load(r filediff.Request)
	// Signal is set, without blocking, after each new update.
	Signal() <-chan struct{}
	// Drain hands over the updates since the last call, oldest first; it
	// computes nothing.
	Drain() []filediff.Update
}
    FileDiffLoader loads the diff of a file in the background. Load never
    blocks; the outcome waits in Drain until the UI collects it, so a slow file
    never holds up the terminal.

type Hoverer interface {
	// Hover asks at p. Only the answer to the last p asked is handed on.
	// In a commit not checked out yet it starts the checkout and answers
	// NotReady.
	Hover(p hover.Position) hover.Submit
	// CancelHover drops the request: no answer comes for it.
	CancelHover(id hover.ReqID)
}
    Hoverer asks the language server about a position. Neither call blocks;
    the answer comes through LSPUpdates.

type LSPUpdates interface {
	// Signal is set, without blocking, after each new update.
	Signal() <-chan struct{}
	// Drain hands over the updates since the last call, oldest first; it
	// computes nothing.
	Drain() []hover.Update
}
    LSPUpdates hands out the answers and status changes of the servers.

type ReleaseNotifier interface {
	// Check asks for the latest release in the background, if asking is on,
	// the running seam is a release and the last check is a day old. It
	// returns at once; seam stopping cancels it.
	Check()
	// Newer is the latest release the last check found, if it is newer
	// than the running one; it asks no one.
	Newer(ctx context.Context) (release.Version, bool)
}
    ReleaseNotifier tells of a newer release of seam.

type StreamLoader interface {
	// Want asks for the files of rs in order, the visible ones first. It
	// replaces the files asked for before; the running load goes on if rs
	// asks for the same file in the same generation.
	Want(rs []filediff.Request)
	// Signal is set, without blocking, after each new update.
	Signal() <-chan struct{}
	// Drain hands over the updates since the last call, oldest first; it
	// computes nothing.
	Drain() []filediff.Update
}
    StreamLoader loads the diffs of the files the stream shows, and of those
    near them, in the background. Like FileDiffLoader it never blocks.

type WatchStarter interface {
	// Watch watches the worktree or the index of c in the background until
	// seam stops; the outcome reaches WatchUpdates. It returns at once.
	Watch(c change.Comparison)
}
    WatchStarter starts watching what a comparison's right side reads from.

type WatchUpdates interface {
	// Signal is set, without blocking, after each new update.
	Signal() <-chan struct{}
	// Drain hands over the updates since the last call, oldest first.
	Drain() []change.WatchUpdate
}
    WatchUpdates hands on what the watcher saw.
