package hover // import "seam/internal/domain/hover"

Package hover holds the values that pass between the viewer and the language
servers: the documents a server is told about, the position a hover asks for,
how a request was taken, and what comes back. It speaks no protocol and starts
no server.

CONSTANTS

const MaxLine = 64 << 10
    MaxLine is the longest line, in bytes, that hovers ask about. Past it the
    text is likely generated or minified, where an answer helps little and
    finding the grapheme and the identifier costs on every move.

const MaxText = 32 << 10
    MaxText is the most of an answer's text, in bytes, that is read: past it the
    rest is cut, which keeps the work per answer small.


VARIABLES

var ErrColourerFailed = errors.New("hover colourer failed")
    ErrColourerFailed is why answers with code come uncoloured once colouring
    them has failed.


TYPES

type Block struct {
	Kind   BlockKind
	Spans  []Span
	Lang   lang.Lang
	Code   string
	Tokens syntax.Tokens
}
    Block is a part of an answer.

type BlockKind uint8
    BlockKind tells which fields of a Block are set.

const (
	// Paragraph sets Spans: text that flows to the width it is shown in.
	Paragraph BlockKind = iota + 1
	// Heading sets Spans, like Paragraph.
	Heading
	// Code sets Code, its language Lang if known, and, once coloured,
	// Tokens; its lines are kept as they are.
	Code
	// Rule is a line across, between parts of the answer.
	Rule
)
type Content struct {
	Blocks []Block
	Cut    bool
}
    Content is an answer as blocks to show one after the other. Cut says the
    answer was longer than MaxText and lost its end.

func (c Content) HasCode() bool
    HasCode says that some block is code in a known language, which a
    highlighter can colour.

type DocKey struct {
	Commit change.ObjectID
	Path   string
}
    DocKey names a document: the file at Path, relative to the repository root
    in slash form, in the worktree if Commit is zero, else as Commit has it,
    served from a checkout of that commit.

type DocVersion uint64
    DocVersion counts the texts a document has had in a session. It rises
    exactly when the text changes, so an answer to an older version can be told
    apart from one to the text on screen.

type Document struct {
	Commit  change.ObjectID
	Path    string
	Lang    lang.Lang
	Version DocVersion
	Text    string
}
    Document is the text a language server is to see for a file: a side of the
    diff on screen.

func (d Document) Key() DocKey
    Key names the document.

type Outcome int
    Outcome is how a hover request ended.

const (
	// Text is an answer with content.
	Text Outcome = iota + 1
	// Empty is an answer without content.
	Empty
	// Error is a request that failed, by the server's answer or its end.
	Error
	// TimedOut is a request the server did not answer in time.
	TimedOut
)
type Position struct {
	Req     ReqID
	Commit  change.ObjectID
	Path    string
	Version DocVersion
	Line    int
	Byte    int
}
    Position is where a hover asks: in the document of Commit and Path as of
    Version, at a byte offset into a line, both counted from zero.

func (p Position) Key() DocKey
    Key names the document asked about.

type ReqID uint64
    ReqID names a hover request. The viewer hands out strictly increasing ones
    before it asks, so an answer finds its request whenever it comes.

type ServerStatus int
    ServerStatus is the state of a server instance.

const (
	Missing ServerStatus = iota + 1
	Starting
	Ready
	ServerFailed
	Stopped
)
    The states of a server instance; see the plan's state machine.

type Span struct {
	Kind SpanKind
	Text string
}
    Span is a stretch of a paragraph. A newline in Text breaks the line there;
    elsewhere a paragraph breaks where it fits.

type SpanKind uint8
    SpanKind is how a stretch of a paragraph stands out. Where one is nested in
    another, the later kind shows: code within a link's text stays code.

const (
	Plain SpanKind = iota
	Emphasis
	Strong
	// Link is the text of a link, URL its target.
	Link
	URL
	// InlineCode is code within the text.
	InlineCode
)
    The kinds of span; Plain is text that does not stand out.

type Submit int
    Submit is how a hover request was taken.

const (
	// Sent means the answer will come as an update.
	Sent Submit = iota + 1
	// NotReady means the server is starting; ask again once it is Ready.
	NotReady
	// Unsupported means no server hovers over the document.
	Unsupported
	// Stale means Version is not the one last synced for the document.
	Stale
	// Busy means the server has too much waiting; ask again later.
	Busy
	// Failed means the server has failed and takes no more requests.
	Failed
)
type Update struct {
	Kind UpdateKind

	Instance lang.Instance
	Commit   change.ObjectID
	Status   ServerStatus

	Req     ReqID
	Version DocVersion
	Outcome Outcome
	// Content is the answer, for Text; its code comes coloured where the
	// viewer can colour it.
	Content Content
}
    Update is news from the language servers.

type UpdateKind int
    UpdateKind tells which fields of an Update are set.

const (
	// StatusChanged sets Instance and Status.
	StatusChanged UpdateKind = iota + 1
	// HoverDone sets Req, Version, Outcome and, for Text, Content.
	HoverDone
	// CheckoutChanged sets Commit and Status, of the checkout that serves
	// Commit's documents: Starting while it is made, Ready once it is,
	// ServerFailed if it could not be.
	CheckoutChanged
)
